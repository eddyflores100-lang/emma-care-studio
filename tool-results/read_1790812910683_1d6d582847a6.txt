    69→    69→function moodOf(rt: PetRuntime): Mood {
    70→    70→  if (rt.sick) return 'triste'
    71→    71→  if (rt.stats.comida < 30) return 'hambre'
    72→    72→  if (rt.stats.descanso < 30) return 'sueno'
    73→    73→  if (rt.stats.felicidad < 25) return 'triste'
    74→    74→  if (rt.stats.felicidad >= 85) return 'feliz'
    75→    75→  return 'normal'
    76→    76→}

   134→   134→/** Aplica el efecto de una regla a una mascota (con partícula y sonido) */
   135→   135→function fireRule(set: StoreSet, get: StoreGet, rule: Rule, pet: WorldObject) {
   136→   136→  const s = get()
   137→   137→  const rt0 = s.pets[pet.id]
   138→   138→  if (!rt0) return
   139→   139→  const rt: PetRuntime = { ...rt0, stats: { ...rt0.stats } }
   140→   140→  const delta = rule.effect === 'aumentar' ? rule.amount : -rule.amount
   141→   141→  rt.stats[rule.stat] = clamp(rt.stats[rule.stat] + delta)
   142→   142→  set({ pets: { ...get().pets, [pet.id]: rt } })
   143→   143→  get().spawnParticles(delta > 0 ? STATS[rule.stat].emoji : '💨', pet.x, pet.y - 3, 1)
   144→   144→  sfx.pop()
   145→   145→}

   147→   147→/** Haz que una mascota "hable": sonido de su especie + globo con la onomatopeya */
   148→   148→function speak(set: StoreSet, get: StoreGet, pet: WorldObject, mood: Mood) {
   149→   149→  petVoice(pet.catalogId, mood)
   150→   150→  const text = VOICES[pet.catalogId]?.[mood]
   151→   151→  if (!text) return
   152→   152→  set({ say: { ...get().say, [pet.id]: { text, until: Date.now() + 2400 } } })
   153→   153→}
